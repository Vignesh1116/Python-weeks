Write down the below functions.

Input is a List L

first(L)
    - return the first element of L

last(L)
    - return the last element of L
init(L)
    - return all but the last element of L
rest(L)
    - return all but the first element of L
member(x, L)
    - return true if x is a member of L

# Return the first element of the list
def first(L):
    if L:  # Check if list is not empty
        return L[0]
    return None  # or raise an error

# Return the last element of the list
def last(L):
    if L:
        return L[-1]
    return None

# Return all but the last element of the list
def init(L):
    if L:
        return L[:-1]
    return []

# Return all but the first element of the list
def rest(L):
    if L:
        return L[1:]
    return []

# Return True if x is a member of the list
def member(x, L):
    return x in L


# Example usage
L = [1, 2, 3, 4, 5]

print("first:", first(L))    # 1
print("last:", last(L))      # 5
print("init:", init(L))      # [1, 2, 3, 4]
print("rest:", rest(L))      # [2, 3, 4, 5]
print("member 3:", member(3, L))  # True
print("member 6:", member(6, L))  # False
